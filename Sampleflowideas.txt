const Phase = ({ num, label, color, width, children }) => (

const FieldChip = ({ value, onChange, onDelete }) => {
  const [editing, setEditing] = useState(false);
  const [val, setVal] = useState(value);
  const ACC = "#06B6D4";
  useEffect(() => setVal(value), [value]);

  if (editing) return (
    <input
      autoFocus value={val}
      onChange={e => setVal(e.target.value)}
      onBlur={() => { setEditing(false); onChange(val); }}
      onKeyDown={e => e.key === "Enter" && setEditing(false)}
      style={{
        background: ACC + "18", border: `1px solid ${ACC}`,
        borderRadius: 5, color: "#E2E8F0", fontSize: 9.5, fontWeight: 600,
        padding: "3px 8px", outline: "none", width: 130, fontFamily: "inherit",
      }}
    />
  );
  return (
    <div style={{
      display: "inline-flex", alignItems: "center", gap: 5,
      background: ACC + "10", border: `1px solid ${ACC}30`,
      borderRadius: 5, padding: "3px 9px",
    }}>
      <span onClick={() => setEditing(true)} style={{ fontSize: 9.5, fontWeight: 600, color: "#CBD5E1", cursor: "text" }} title="Click to rename">
        {val}
      </span>
      <span onClick={onDelete} style={{ fontSize: 8, color: "#475569", cursor: "pointer" }} title="Remove">✕</span>
    </div>
  );
};
